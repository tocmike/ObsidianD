VAR1A:=(HHV(HIGH,4)-CLOSE)/(HHV(HIGH,4)-LLV(LOW,4))*100-90;

{ --- 2. 定义柱子状态与长度 --- }
今日红柱 := REF(砖型图,1) < 砖型图;          { 今天数值大于昨天，表现为红柱 }
昨日绿柱 := REF(砖型图,2) > REF(砖型图,1);    { 昨天数值小于前天，表现为绿柱 }

昨日绿柱长度 := REF(砖型图,2) - REF(砖型图,1); { 昨天跌幅构成的绿柱绝对长度 }

{ --- 3. 原有选股条件 --- }
条件1 := 今日红柱 AND 昨日绿柱;               { 满足：昨天是绿砖，今天是红砖 }
条件2 := 红柱长度 > 昨日绿柱长度 * 0.5;       { 满足：今天的红砖长度 > 昨天绿砖的一半 }

{ --- 4. 新增：3天横盘后强势突破条件（对应图片形态） --- }
{ 计算过去3天（不含今日）的最高价和最低价 }
三日最高 := REF(HHV(HIGH, 3), 1); 
三日最低 := REF(LLV(LOW, 3), 1);

{ 判定横盘：前3天的整体振幅极小，这里设定为小于5%，代表市场在此区间高度共识 }
横盘振幅 := (三日最高 - 三日最低) / 三日最低 * 100;
条件3 := 横盘振幅 < 5; 

{ 判定突破：今天收盘价突破前3天最高点，且是一根明显的大阳线 }
今日涨幅 := (CLOSE - REF(CLOSE, 1)) / REF(CLOSE, 1) * 100;
条件4 := CLOSE > 三日最高 AND CLOSE > OPEN AND 今日涨幅 > 4; 

{ --- 5. 综合输出选股信号 --- }
XG: 条件1 AND 条件2 AND 条件3 AND 条件4;
